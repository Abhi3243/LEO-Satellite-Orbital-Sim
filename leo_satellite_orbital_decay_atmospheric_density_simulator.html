Why Solar Activity Accelerates Orbital Decay

1. Aerodynamic Drag Force Equation

F_D = ½ · ρ(h) · C_d · A · v²
  • ρ(h): Atmospheric density at orbital altitude h [kg/m³]
  • C_d: Drag coefficient in free-molecular flow (~2.2 for satellites)
  • A: Frontal cross-sectional area facing the relative velocity [m²]
  • v: Orbital velocity relative to the atmosphere [m/s]

2. Circular Orbital Velocity & Energy

v = √(μ / r)  |  E_spec = - μ / (2r)
  • μ: Earth's standard gravitational parameter = 3.986004418 × 10¹⁴ m³/s²
  • r: Orbital radius = R_Earth + h (where R_Earth = 6,371 km)
  • E_spec: Specific mechanical orbital energy (kinetic + gravitational)

3. King-Hele Altitude Loss Rate (dh/dt)

dh/dt = - √(μ · r) · (C_d · A / m) · ρ(h)

Derived by equating rate of energy dissipation from aerodynamic drag (dE/dt = - F_D · v) to the rate of potential energy change with altitude (dE/dr = μm / 2r²). As atmospheric density increases exponentially, altitude decay accelerates dramatically.

4. Segmented Exponential Density (NRLMSIS-00)

ρ(h) = ρ_i · exp[ - (h - h_i) / H_i ] · S_mult

Standard segmented atmospheric model matching NASA/NOAA upper thermosphere reference data. Density drops by an order of magnitude every ~50 km. The solar activity multiplier (S_mult) simulates thermospheric expansion during high solar radio flux (F10.7) and geomagnetic storms.

# LEO SATELLITE ORBITAL DECAY SIMULATOR (PROTOTYPE)
# State Science and Engineering Fair (SSEF) - Physics & Astronomy (Mechanics)
# Guiding Question:
# "How does atmospheric density at different altitudes affect the predicted
#  orbital decay rate and lifetime of low-Earth-orbit satellites?"
# ==============================================================================

import numpy as np
import matplotlib.pyplot as plt
import math

# ------------------------------------------------------------------------------
# 1. PHYSICAL CONSTANTS & REFERENCE DATA
# ------------------------------------------------------------------------------
MU = 3.986004418e14   # Earth standard gravitational parameter (m^3 / s^2)
R_EARTH = 6371.0e3    # Mean Earth radius (meters)

# Segmented Exponential Atmosphere Model (US Standard / NRLMSIS-00 reference)
# Format: (base_altitude_km, base_density_kg_m3, scale_height_km)
ATMOSPHERE_TABLE = [
    (100, 5.297e-7,  5.877),
    (120, 2.438e-8,  7.263),
    (150, 2.070e-9,  26.4),
    (200, 2.541e-10, 37.5),
    (250, 6.073e-11, 44.8),
    (300, 1.916e-11, 50.3),
    (350, 7.014e-12, 54.8),
    (400, 2.801e-12, 58.2),
    (450, 1.184e-12, 61.3),
    (500, 5.215e-13, 64.5),
    (600, 1.137e-13, 71.7),
    (700, 3.070e-14, 80.2),
    (800, 1.136e-14, 92.5),
    (900, 5.759e-15, 111.0),
    (1000, 3.561e-15, 140.0)
]

def get_atmospheric_density(altitude_km, density_multiplier=1.0):
    """Calculates atmospheric density (kg/m^3) at a given altitude in km."""
    if altitude_km < 100:
        altitude_km = 100
    if altitude_km >= ATMOSPHERE_TABLE[-1][0]:
        h0, rho0, H = ATMOSPHERE_TABLE[-1]
        return rho0 * math.exp(-(altitude_km - h0) / H) * density_multiplier
    for i in range(len(ATMOSPHERE_TABLE) - 1):
        if ATMOSPHERE_TABLE[i][0] <= altitude_km < ATMOSPHERE_TABLE[i+1][0]:
            h0, rho0, H = ATMOSPHERE_TABLE[i]
            return rho0 * math.exp(-(altitude_km - h0) / H) * density_multiplier
    return ATMOSPHERE_TABLE[0][1] * density_multiplier

def get_orbital_velocity(altitude_m):
    """Calculates circular orbital velocity (m/s)."""
    return math.sqrt(MU / (R_EARTH + altitude_m))

def get_drag_force(altitude_km, mass_kg=500.0, area_m2=2.0, cd=2.2, density_multiplier=1.0):
    """Calculates aerodynamic drag force (Newtons)."""
    rho = get_atmospheric_density(altitude_km, density_multiplier)
    v = get_orbital_velocity(altitude_km * 1000.0)
    return 0.5 * rho * cd * area_m2 * (v ** 2)

def dh_dt(altitude_m, mass_kg=500.0, area_m2=2.0, cd=2.2, density_multiplier=1.0):
    """Calculates rate of altitude loss (m/s) using King-Hele drag formulation."""
    r = R_EARTH + altitude_m
    rho = get_atmospheric_density(altitude_m / 1000.0, density_multiplier)
    ballistic_inv = (cd * area_m2) / mass_kg
    return - math.sqrt(MU * r) * ballistic_inv * rho

def simulate_orbital_decay(h_start_km, mass_kg=500.0, area_m2=2.0, cd=2.2,
                           density_multiplier=1.0, cutoff_km=120.0, max_days=3650.0):
    """
    Simulates altitude decay over time using Runge-Kutta 4th order (RK4).
    Returns (times_days, altitudes_km).
    """
    t_sec = 0.0
    h_m = h_start_km * 1000.0
    times = [0.0]
    alts = [h_start_km]
    max_sec = max_days * 86400.0

    while h_m > (cutoff_km * 1000.0) and t_sec < max_sec:
        rate = abs(dh_dt(h_m, mass_kg, area_m2, cd, density_multiplier))
        if rate < 1e-12:
            break
        # Adaptive time-step
        dt = min(86400.0 * 5.0, 1000.0 / rate)
        dt = max(dt, 10.0)

        # RK4 Integration
        k1 = dh_dt(h_m, mass_kg, area_m2, cd, density_multiplier)
        k2 = dh_dt(h_m + 0.5 * dt * k1, mass_kg, area_m2, cd, density_multiplier)
        k3 = dh_dt(h_m + 0.5 * dt * k2, mass_kg, area_m2, cd, density_multiplier)
        k4 = dh_dt(h_m + dt * k3, mass_kg, area_m2, cd, density_multiplier)

        dh = (k1 + 2*k2 + 2*k3 + k4) / 6.0 * dt
        h_m += dh
        t_sec += dt

        times.append(t_sec / 86400.0)
        alts.append(h_m / 1000.0)

    return np.array(times), np.array(alts)

# 2. RUN SCIENCE FAIR EXPERIMENTAL TRIALS
# ------------------------------------------------------------------------------
print("=" * 70)
print("LEO SATELLITE ORBITAL DECAY SIMULATOR: SCIENCE FAIR TRIALS")
print("=" * 70)

# Trial Set 1: Effect of Starting Altitude
altitudes = [250, 300, 350, 400, 450]
plt.figure(figsize=(10, 6))
print("
--- TRIAL SET 1: EFFECT OF STARTING ALTITUDE ---")
for alt in altitudes:
    t, h = simulate_orbital_decay(alt, mass_kg=500, area_m2=2.0, cd=2.2, density_multiplier=1.0)
    init_drag = get_drag_force(alt, mass_kg=500, area_m2=2.0, cd=2.2) * 1000.0 # mN
    lifetime = t[-1]
    print(f"Alt {alt} km | Initial Drag: {init_drag:8.3f} mN | Lifetime: {lifetime:7.1f} days ({lifetime/365.25:5.2f} yrs)")
    plt.plot(t, h, label=f"h0 = {alt} km (Lifetime: {lifetime:.0f} d)")

plt.title("Effect of Starting Altitude on Orbital Decay", fontsize=14, fontweight='bold')
plt.xlabel("Time (Days)", fontsize=11)
plt.ylabel("Orbital Altitude (km)", fontsize=11)
plt.grid(True, linestyle='--', alpha=0.6)
plt.legend()
plt.tight_layout()
plt.show()

# Trial Set 2: Effect of Atmospheric Density Multiplier (Solar Activity)
multipliers = [0.5, 1.0, 2.0, 3.0]
plt.figure(figsize=(10, 6))
print("
--- TRIAL SET 2: EFFECT OF SOLAR ACTIVITY MULTIPLIER (at 350 km) ---")
for mult in multipliers:
    t, h = simulate_orbital_decay(350, mass_kg=500, area_m2=2.0, cd=2.2, density_multiplier=mult)
    init_drag = get_drag_force(350, mass_kg=500, area_m2=2.0, cd=2.2, density_multiplier=mult) * 1000.0
    lifetime = t[-1]
    print(f"Multiplier {mult:.1f}x | Initial Drag: {init_drag:8.3f} mN | Lifetime: {lifetime:7.1f} days ({lifetime/365.25:5.2f} yrs)")
    plt.plot(t, h, label=f"Density {mult:.1f}x (Lifetime: {lifetime:.0f} d)")

plt.title("Effect of Solar Activity (Density Multiplier) on Orbital Decay at 350 km", fontsize=14, fontweight='bold')
plt.xlabel("Time (Days)", fontsize=11)
plt.ylabel("Orbital Altitude (km)", fontsize=11)
plt.grid(True, linestyle='--', alpha=0.6)
plt.legend()
plt.tight_layout()
plt.show()
